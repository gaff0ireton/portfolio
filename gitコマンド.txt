diffzip () {
    # 1. 引数が足りない場合に使い方を表示
    if [ $# -lt 3 ]; then
        echo "Usage: diffzip <from-commit> <to-commit> <output-file>"
        return 1
    fi

    local FROM=$1
    local TO=$2
    local OUT=$3

    # 2. 差分ファイルリストを一度変数に格納
    local DIFF_FILES=$(git diff --name-only "$FROM" "$TO" --diff-filter=ACMR | grep -vE '\.(sass|scss)$')

    # 3. 差分がない場合にエラーを防ぐ
    if [ -z "$DIFF_FILES" ]; then
        echo "No files to archive."
        return 0
    fi

    # 4. アーカイブ実行
    git archive "$TO" $DIFF_FILES -o "$OUT"
    echo "Saved diff to $OUT"
}

distzip () {
    if [ $# -lt 3 ]; then
        echo "Usage: distzip <from-commit> <to-commit> <output-file>"
        return 1
    fi

    local FROM=$1
    local TO=$2
    local OUT=$3

    # 1. ヌル文字（\0）区切りで取得（スペースや改行を含むファイル名対策）
    # 2. dist/ 始まりを抽出し、パスを削る
    # 3. 配列 FILES に格納（Bash 3系でも動く互換性重視の書き方）
    local FILES=()
    while IFS= read -r -d $'\0' file; do
        FILES+=("$file")
    done < <(git diff -z --name-only "$FROM" "$TO" --diff-filter=ACMR | grep -z '^dist/' | sed -z 's|^dist/||')

    if [ ${#FILES[@]} -eq 0 ]; then
        echo "No files found in 'dist/' directory."
        return 0
    fi

    # 4. git archive 実行
    # -- よりも後に配列を展開するのが Git archive の作法
    if git archive -o "$OUT" "$TO:dist" -- "${FILES[@]}"; then
        echo "Successfully saved 'dist/' contents to $OUT"
    else
        echo "Error: Failed to create archive."
        return 1
    fi
}
